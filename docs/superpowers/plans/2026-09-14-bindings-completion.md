> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Finish C/Python/Java/JS bindings so every language scores `bench/pmml/DecisionTreeIris.pmml` through the same `PmmlApi v1` core with tests green.

**Architecture:** Rust owns everything (`Session::run(&dyn Batch)` in `crates/pmmlruntime/src/session/session.rs:441`). C exposes it as versioned `PmmlApi` table (`crates/pmmlruntime/src/ffi/mod.rs`, `include/pmml_runtime.h`). Python/Java/JS are thin shims over `PmmlGetApi(1)` — no direct `use pmmlruntime::Session` outside `ffi` + `python/_native` transition. Arrow uses Arrow C Data Interface everywhere.

**Tech Stack:** Rust 1.78, `cbindgen`, `pyo3 0.22 + maturin`, `jni 0.21 + Maven`, `napi 2 + wasm-bindgen 0.2`, `arrow 53`, `pytest`, `JUnit4`, `node --test`.

```rust
    if sess.is_null() || binding.is_null() { return status_invalid_arg("RunWithBinding: null"); }
    let h = unsafe { &mut *(sess as *mut SessionHandle) };
    let b = unsafe { &mut *(binding as *mut IoBindingHandle) };
    use crate::session::batch::Batch;
    let map = b.inputs.clone();
    let result = match h.session.run(&map as &dyn crate::session::batch::Batch) {
        Ok(r) => r,
        Err(e) => return status_from_error(e),
    };
    b.last = Some(result.into_rows());
    ptr::null_mut()
}
```

```rust
unsafe extern "C" fn api_CopyBindingOutputsToCpu(binding: *mut PmmlIoBinding, out_flat: *mut PmmlValue, out_count: *mut usize) -> *mut PmmlStatus {
    if binding.is_null() || out_flat.is_null() || out_count.is_null() { return status_invalid_arg("CopyBindingOutputsToCpu: null"); }
    let b = unsafe { &*(binding as *mut IoBindingHandle) };
    let rows = match &b.last { Some(r) => r, None => return status_invalid_arg("CopyBindingOutputsToCpu: no RunWithBinding yet") };
    let want: Vec<String> = if b.outputs.is_empty() {
        rows[0].keys().cloned().collect()
    } else { b.outputs.clone() };
    let capacity = unsafe { *out_count };
    if capacity < want.len() { return status_invalid_arg("CopyBindingOutputsToCpu: out buffer too small"); }
    for (i, name) in want.iter().enumerate() {
        let v = rows[0].get(name).copied().unwrap_or(Value::Missing);
        unsafe { *out_flat.add(i) = value_to_pmml_value(v) };
    }
    unsafe { *out_count = want.len() };
    ptr::null_mut()
}
```

```rust
// javascript/node/src/lib.rs
use napi_derive::napi;
use std::collections::HashMap;
use pmmlruntime::base::Value;
use pmmlruntime::session::batch::Batch;
use pmmlruntime::session::{PmmlEnv, Session, SessionOptions};

#[napi]
pub fn hello() -> String { "pmml-runtime".to_string() }

#[napi]
pub struct InferenceSession { sess: Session, _env: PmmlEnv }

#[napi]
impl InferenceSession {
    #[napi(constructor)]
    pub fn new(path: String) -> napi::Result<Self> {
        let env = PmmlEnv::new();
        let sess = Session::from_file(&env, &path, SessionOptions::default())
            .map_err(|e| napi::Error::from_reason(e.to_string()))?;
        Ok(Self { sess, _env: env })
    }
    #[napi]
    pub fn run(&self, input: HashMap<String, f64>) -> napi::Result<HashMap<String, String>> {
        let map: HashMap<String, Value> = input.into_iter().map(|(k, v)| (k, Value::Continuous(v))).collect();
        let out = self.sess.run(&map as &dyn Batch).map_err(|e| napi::Error::from_reason(e.to_string()))?
            .into_single().unwrap();
        Ok(out.into_iter().map(|(k, v)| (k, format!("{:?}", v))).collect())
    }
}
```
